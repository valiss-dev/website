{{/* Article prev/next pager. Override of Hextra v0.12.3 components/pager.html. The theme threads prev/next with .Prev/.NextInSection, which only sees regular-page siblings and so breaks the docs tour into disconnected islands (Concepts and Guides section indexes and their children never link back to the trunk). This walks the full docs reading order instead (layouts/_partials/docs-nav-order.html, the same depth-first weight order as the sidebar), so Next flows Quickstart -> Concepts -> ... -> Guides -> ... -> Reference -> ... straight through. Per-page prev/next front matter still overrides, and prev/next: false still disables a side. */}} {{- $order := partial "docs-nav-order.html" (dict "section" .FirstSection) -}} {{- $prev := false -}} {{- $next := false -}} {{- $idx := -1 -}} {{- range $i, $p := $order -}} {{- if eq $p $ -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if ge $idx 0 -}} {{- if gt $idx 0 -}}{{- $prev = index $order (sub $idx 1) -}}{{- end -}} {{- if lt $idx (sub (len $order) 1) -}}{{- $next = index $order (add $idx 1) -}}{{- end -}} {{- end -}} {{- /* Explicit front-matter overrides win over the computed order. */ -}} {{- if eq .Params.prev false }}{{ $prev = false }}{{ else }}{{- with .Params.prev }}{{- with $.Site.GetPage . }}{{ $prev = . }}{{- end }}{{- end }}{{- end -}} {{- if eq .Params.next false }}{{ $next = false }}{{ else }}{{- with .Params.next }}{{- with $.Site.GetPage . }}{{ $next = . }}{{- end }}{{- end }}{{- end -}} {{- if or $prev $next -}}
{{- if $prev -}} {{- $linkTitle := partial "utils/title" $prev -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:ltr:rotate-180\"") -}} {{- $linkTitle -}} {{- end -}} {{- if $next -}} {{- $linkTitle := partial "utils/title" $next -}} {{- $linkTitle -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx:inline hx:h-5 hx:shrink-0 hx:rtl:-rotate-180\"") -}} {{- end -}}
{{- end -}}
